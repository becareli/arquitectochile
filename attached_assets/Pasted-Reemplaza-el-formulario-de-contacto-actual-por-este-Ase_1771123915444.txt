  let userData = {};

  function setEntry(tipo) {
    userData.categoria = tipo;
    const select = document.getElementById('sub-servicio');
    select.innerHTML = servicios[tipo].map(s => `<option value="${s}">${s}</option>`).join('');
    go(2);
  }

  function go(s) {
    document.querySelectorAll('.step').forEach(d => d.style.display = 'none');
    document.getElementById('step' + s).style.display = 'block';
    document.getElementById('progress').style.width = (s * 20) + '%';
  }

  document.getElementById('architectForm').onsubmit = async (e) => {
    e.preventDefault();
    const finalData = {
      ...userData,

      tipo: document.getElementById('tipo_propiedad').value,
      analisis_sentimiento: document.getElementById('historia').value,
      presupuesto: document.getElementById('presupuesto').value,
      nombre: document.getElementById('nombre').value,
      whatsapp: document.getElementById('whatsapp').value
    };
    
    // Aquí se conecta con n8n
    console.log("Enviando a n8n...", finalData);
    alert("Recibido. Nuestro sistema analizará la factibilidad de tu dirección y te contactaremos por WhatsApp.");
  };
</script>
