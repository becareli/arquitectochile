        let leadData = { branch: '', service: '', budget: '' };

        const services = {
            empresa: [
                {n: 'Revisoría Indep.', i: '⚖️'}, {n: 'ITO Obras', i: '🏗️'}, 
                {n: 'Proy. Eléctricos', i: '⚡'}, {n: 'Proyectos Gas', i: '🔥'}, 
                {n: 'Subdivisión', i: '🗺️'}, {n: 'Loteos', i: '📐'}, {n: 'Diseño 3D', i: '🖥️'}
            ],
            particular: [
                {n: 'Asesoría Terreno', i: '📍'}, {n: 'Ampliaciones', i: '🏠'}, 
                {n: 'Remodelación', i: '✨'}, {n: 'Ley del Mono', i: '📜'}, 
                {n: 'Fusión/Subdiv.', i: '🗺️'}, {n: 'Permisos/3D', i: '📐'}, 
                {n: 'Revisor Indep.', i: '⚖️'}, {n: 'Agua/Alcant.', i: '💧'}, 
                {n: 'Electricidad/Gas', i: '⚡'}, {n: 'Proyecto desde Cero', i: '🌱'}
            ]
        };

        function selectBranch(b) {
            leadData.branch = b;
            document.getElementById('step-1').classList.add('hidden');
            document.getElementById('step-2').classList.remove('hidden');
            document.getElementById('speech-bubble').innerHTML = b === 'empresa' ? 
                "Entendido. Somos expertos en normativa corporativa. <b>¿Qué servicio requiere la empresa?</b>" : 
                "Excelente. Ayudamos a familias a construir sueños legalmente. <b>¿Qué necesitas hacer?</b>";
            
            const grid = document.getElementById('services-grid');
            grid.innerHTML = services[b].map(s => `
                <button onclick="selectService('${s.n}')" class="p-3 border rounded-xl flex flex-col items-center text-[10px] font-bold hover:bg-gray-50">
                    <span class="text-xl mb-1">${s.i}</span> ${s.n}
                </button>
            `).join('') + `<button onclick="selectService('Especial')" class="col-span-2 p-3 border-2 border-dashed rounded-xl text-gray-400 text-xs font-bold uppercase mt-2">¿No encuentras lo que buscas?</button>`;
        }

        function selectService(s) {
            leadData.service = s;
            if (s === 'Proyecto desde Cero') {
                showStep('step-budget');
            } else {
                showStep('step-form');
            }
        }

        function setBudget(b) { leadData.budget = b; showStep('step-form'); }

        function showStep(id) {
            document.querySelectorAll('#app-container > div:not(:first-child)').forEach(el => el.classList.add('hidden'));
            document.getElementById(id).classList.remove('hidden');
        }

        function submitLead() {
            if(!document.getElementById('calle').value || !document.getElementById('comuna').value) return alert("Calle y Comuna son obligatorios");
            showStep('step-thanks');
            document.getElementById('avatar').style.display = 'none';
            document.getElementById('speech-bubble').innerHTML = "<b>¡Solicitud enviada con éxito!</b>";
        }

        function reset() { location.reload(); }
    </script>
