<script>
    let areas = [];
    let leadSvc = "";

    document.querySelectorAll('.choice-btn').forEach(b => {
        b.onclick = function() {
            this.classList.toggle('selected');
            const c = this.querySelector('input');
            c.checked = !c.checked;
        }
    });

    function nav(n) {
        document.querySelectorAll('.step').forEach(s => s.classList.remove('active'));
        document.getElementById('step'+n).classList.add('active');
    }

    function goTo2() {
        const checked = document.querySelectorAll('input[name="area"]:checked');
        if(checked.length === 0) return alert("Por favor, seleccione un área.");
        areas = Array.from(checked).map(i => i.value);
        nav(2);
    }

    function updateFiles() {
        const f = document.getElementById('archivos').files;
        document.getElementById('file-status').innerText = f.length + " archivos seleccionados ✅";
    }

    function evalGift() {
        // Lógica de Prioridad de Regalo
        if(areas.includes("Construcción")) leadSvc = "Construcción";
        else if(areas.includes("Regularización")) leadSvc = "Regularización";
        else if(areas.includes("Terrenos")) leadSvc = "Terrenos";
        else leadSvc = "Ingeniería";

        const titles = {
            "Construcción": "Manual: Construir sin Sobrecostos",
            "Regularización": "Guía: Ley del Mono 2026",
            "Terrenos": "Manual: Subdivisión de Terrenos",
            "Ingeniería": "Checklist: Proyectos Eléctricos"
        };

        document.getElementById('gift-text').innerHTML = `Excelente. Como le interesa la <b>${leadSvc}</b>, Patricio le obsequiará su material:`;
        document.getElementById('gift-title').innerText = titles[leadSvc];
        nav(4);
    }

    document.getElementById('advisorForm').onsubmit = function(e) {
        e.preventDefault();
        alert("Antecedentes enviados. El sistema de Patricio Vega ha priorizado su revisión.");
        location.reload();
    };
</script>
