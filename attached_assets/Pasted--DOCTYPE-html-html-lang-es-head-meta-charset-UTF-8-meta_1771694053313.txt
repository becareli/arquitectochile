            leadData.propertyType = type;
            document.querySelectorAll('.prop-type-btn').forEach(b => b.classList.remove('bg-blue-600', 'text-white'));
            event.target.classList.add('bg-blue-600', 'text-white');
        }

        function showStep(id) {
            document.querySelectorAll('#app-container > div:not(:first-child)').forEach(el => el.classList.add('hidden'));
            document.getElementById(id).classList.remove('hidden');
        }

            if(!document.getElementById('direccion').value || !document.getElementById('comuna').value) {
